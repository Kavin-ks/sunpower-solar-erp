    %% ========================================================
    %% TABLE SCHEMAS (CORE ATTRIBUTES)
    %% ========================================================
    CUSTOMERS {
        bigint id PK
        varchar full_name
        varchar phone
        varchar email
        varchar city
        varchar customer_type
    }
    ENQUIRIES {
        bigint id PK
        varchar enquiry_number UK
        bigint customer_id FK
        decimal requested_capacity_kw
        varchar system_type
        varchar status
        bigint assigned_to_user_id FK
    }
    SITE_SURVEYS {
        bigint id PK
        varchar survey_number UK
        bigint customer_id FK
        bigint enquiry_id FK
        bigint technician_id FK
        decimal available_roof_area_sq_ft
        varchar status
        varchar approved_by
    }
    SOLAR_PACKAGES {
        bigint id PK
        varchar name
        varchar system_type
        decimal capacity_kw
        decimal base_price
    }
    PACKAGE_BOMS {
        bigint id PK
        bigint package_id FK
        bigint product_id FK
        int quantity
    }
    PRODUCTS {
        bigint id PK
        varchar product_code UK
        varchar name
        bigint category_id FK
        decimal purchase_price
        decimal selling_price
        int current_stock
        boolean serialized
    }
    QUOTATIONS {
        bigint id PK
        varchar quotation_number UK
        bigint customer_id FK
        bigint site_survey_id FK
        bigint solar_package_id FK
        decimal subtotal
        decimal tax_amount
        decimal total_amount
        varchar status
    }
    SALES_ORDERS {
        bigint id PK
        varchar order_number UK
        bigint customer_id FK
        bigint quotation_id FK
        decimal total_amount
        decimal advance_required
        decimal total_paid
        decimal remaining_balance
        varchar status
    }
    PAYMENTS {
        bigint id PK
        varchar payment_number UK
        bigint order_id FK
        bigint customer_id FK
        decimal amount
        varchar method
        varchar reference_number
        varchar status
    }
    INVENTORY_STOCKS {
        bigint id PK
        bigint product_id FK
        int available_quantity
        int reserved_quantity
        varchar warehouse_location
    }
    DISPATCHES {
        bigint id PK
        varchar dispatch_number UK
        bigint sales_order_id FK
        bigint customer_id FK
        varchar status
    }
    SERIAL_NUMBERS {
        bigint id PK
        varchar serial_code UK
        bigint product_id FK
        varchar status
        bigint sales_order_id FK
        bigint customer_id FK
    }
    INSTALLATIONS {
        bigint id PK
        varchar installation_number UK
        bigint sales_order_id FK
        bigint customer_id FK
        boolean testing_passed
        varchar customer_signature_url
        varchar status
    }
    WARRANTIES {
        bigint id PK
        varchar warranty_number UK
        bigint serial_number_id FK
        bigint customer_id FK
        bigint sales_order_id FK
        date warranty_start_date
        date warranty_end_date
        varchar status
    }
    SERVICE_TICKETS {
        bigint id PK
        varchar ticket_number UK
        bigint customer_id FK
        bigint serial_number_id FK
        bigint warranty_id FK
        varchar status
        boolean under_warranty
    }
    INVOICES {
        bigint id PK
        varchar invoice_number UK
        bigint sales_order_id FK
        bigint customer_id FK
        decimal total_amount
        decimal paid_amount
        decimal balance_amount
        varchar status
    }
